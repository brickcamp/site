{{/* Returns all entries sorted by a field. `from` says where the field lives: "page" (the default) is a field on the page itself, "facts" a dotted path into _partials/entries/facts. Params: one option from data/entries/sortings.toml. */}} {{- $field := index . "field" -}} {{- $order := index . "order" | default "asc" -}} {{- $from := index . "from" | default "page" -}} {{- $slug := index . "slug" -}} {{- $result := slice -}} {{- if eq $from "facts" -}} {{- $map := slice -}} {{- range partial "entries/getAll" -}} {{- $value := partial "entries/facts" . -}} {{- range split $field "." -}} {{- $value = index $value . -}} {{- end -}} {{- if eq $value nil -}} {{- errorf "sorting %q: no fact %q" $slug $field -}} {{- end -}} {{- $map = $map | append (dict "key" . "val" $value) -}} {{- end -}} {{- range sort $map "val" $order -}} {{- $result = $result | append .key -}} {{- end -}} {{- else -}} {{- $entries := partial "entries/getAll" -}} {{- $result = sort $entries $field $order -}} {{- end -}} {{- return $result -}}
