{{/* Returns an entry's tag values that fall within a filter's scope, joined with "|" (e.g. "28|45"). The scope mirrors the type dimension of the filter: a listed type, or any type. Empty for filters without a base (the unfiltered lookup). */}} {{- $page := .page -}} {{- $filter := .filter -}} {{- $base := index $filter "base" | default false -}} {{- $type := index $filter "type" | default false -}} {{- $values := slice -}} {{- with $base -}} {{- range $tag := ($page.Params.tags | default slice) -}} {{- $seg := partial "tags/getSegments" $tag -}} {{- if ne $seg.base $base -}} {{- continue -}} {{- end -}} {{- with $type -}} {{- if ne . $seg.type -}} {{- continue -}} {{- end -}} {{- end -}} {{- if $seg.value -}} {{- $values = $values | append (string $seg.value) -}} {{- end -}} {{- end -}} {{- end -}} {{- return delimit (uniq $values) "|" -}}
