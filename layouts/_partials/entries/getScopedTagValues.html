{{/* Returns an entry's tag values that fall within a filter's scope, joined with "|" (e.g. "28|45"). The scope mirrors the type dimension of the filter: a listed type, or any type. Empty for filters without a base (the unfiltered lookup). The match rule lives in tags/matchesFilter. Params: page, filter. */}} {{- $page := .page -}} {{- $filter := .filter -}} {{- $values := slice -}} {{- with index $filter "base" | default false -}} {{- range $tag := ($page.Params.tags | default slice) -}} {{- $seg := partial "tags/getSegments" $tag -}} {{- if not (partial "tags/matchesFilter" (dict "filter" $filter "segments" $seg)) -}} {{- continue -}} {{- end -}} {{- if $seg.value -}} {{- $values = $values | append (string $seg.value) -}} {{- end -}} {{- end -}} {{- end -}} {{- return delimit (uniq $values) "|" -}}
