{{/* Returns all entries whose tags fit the provided filter. Walks the tag taxonomy once and keeps the pages of every tag that passes. Filters on base and type only — never on value. Entries carry values no dropdown lists, so value filtering happens in the client (see lookup.js). Params: base, type — each optional; an empty filter passes every tag. */}} {{- $hasBase := index . "base" | default false -}} {{- $hasType := index . "type" | default false -}} {{- $result := slice -}} {{- range $rawTag, $weightedPages := site.Taxonomies.tags -}} {{- $tag := partial "tags/getSegments" $rawTag -}} {{- with $hasBase -}} {{- if ne . $tag.base -}} {{- continue -}} {{- end -}} {{- end -}} {{- with $hasType -}} {{- if ne . $tag.type -}} {{- continue -}} {{- end -}} {{- end -}} {{- range $weightedPages }} {{- $result = $result | append .Page -}} {{- end -}} {{- end -}} {{- return uniq $result -}}
