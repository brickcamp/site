{{/* Returns all entries whose tags fit the provided filter. Walks the tag taxonomy once and keeps the pages of every tag that passes. The filter's shape and the match rule live in tags/matchesFilter. Params: a filter. */}} {{- $filter := . -}} {{- $result := slice -}} {{- range $rawTag, $weightedPages := site.Taxonomies.tags -}} {{- $seg := partial "tags/getSegments" $rawTag -}} {{- if not (partial "tags/matchesFilter" (dict "filter" $filter "segments" $seg)) -}} {{- continue -}} {{- end -}} {{- range $weightedPages }} {{- $result = $result | append .Page -}} {{- end -}} {{- end -}} {{- return uniq $result -}}
