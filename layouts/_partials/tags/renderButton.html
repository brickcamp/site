{{/* Renders one tag as a pill button linking into the search. Params: the segments dict from tags/getSegments. */}} {{- $seg := . -}} {{/* renderMetadata gates on hasScope, which reads the same filters the catalog does — so a base that has a scope but no scope entry is a broken build. errorf does not halt Hugo, so the empty scope keeps the rest running on raw slugs rather than piling a nil dereference on top of the message. */}} {{- $scope := dict "types" slice "values" slice "valueFormat" "" -}} {{- $found := false -}} {{- range (partialCached "scopes/getCatalog" .).scopes -}} {{- if eq .slug $seg.base -}} {{- $scope = . -}} {{- $found = true -}} {{- end -}} {{- end -}} {{- if not $found -}} {{- errorf "tag base %q has a search scope but no scope catalog entry" $seg.base -}} {{- end -}} {{/* Empty when the tag names no type, which is a base with nothing narrowing it — the search opens on the whole scope. */}} {{- $typeTitle := $seg.type -}} {{- with where $scope.types "slug" $seg.type -}} {{- $typeTitle = (index . 0).title -}} {{- end -}} {{- $valueTitle := "" -}} {{- with $seg.value -}} {{- $slug := string . -}} {{- $match := where $scope.values "slug" $slug -}} {{- if $match -}} {{- $valueTitle = (index $match 0).title -}} {{- else if $scope.valueFormat -}} {{- $valueTitle = printf $scope.valueFormat $slug -}} {{- else -}} {{- $valueTitle = $slug -}} {{- end -}} {{- end -}} {{/* The label has one colon slot, filled only when there is something to fill it with. The value takes it where there is one; otherwise the type does, since a bare "Circle" or "Linear" says too little on its own. A tag naming no type has nothing to put in the slot at all. The hover text spells out however many of the three the tag actually names. */}} {{- $baseTitle := title $seg.base -}} {{- $label := $baseTitle -}} {{- $hover := $baseTitle -}} {{- with $typeTitle -}} {{- $label = printf "%s: %s" $baseTitle . -}} {{- $hover = printf "%s of type %s" $baseTitle . -}} {{- end -}} {{- with $valueTitle -}} {{- $label = printf "%s: %s" (or $typeTitle $baseTitle) . -}} {{- $hover = printf "%s: %s" $hover . -}} {{- end -}} {{- $link := printf "/?base=%s" $seg.base -}} {{- with $seg.type -}} {{- $link = printf "%s&type=%s" $link . -}} {{- end -}} {{- with $seg.value -}} {{- $link = printf "%s&value=%d" $link . -}} {{- end -}} {{ $label }}
