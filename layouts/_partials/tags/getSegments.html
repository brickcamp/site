{{/* Canonical tag grammar: tags are "base-type-value" (e.g. "angle-studturn-28"). Splits a tag into those three segments, defaulting a missing type to "else" and a missing value to 0. Also accepts the .Path of a term page as input. */}} {{- $tag := strings.TrimPrefix "/tags/" . -}} {{- $split := split $tag "-" -}} {{- $base := index $split 0 -}} {{- $type := index $split 1 | default "else" -}} {{- $value := index $split 2 | default 0 -}} {{- return dict "base" $base "type" $type "value" $value -}}
