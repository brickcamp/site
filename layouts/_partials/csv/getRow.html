{{/* Returns one lookup row: the given fields, tab-separated. Read back by the row parsers in assets/js/home/lookup.js Params: the fields, as a slice. */}} {{- $fields := slice -}} {{- range . -}} {{- $fields = $fields | append (replaceRE "[\t\r\n]+" " " (string .)) -}} {{- end -}} {{- return delimit $fields "\t" -}}
